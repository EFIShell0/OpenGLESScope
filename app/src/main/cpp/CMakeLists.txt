cmake_minimum_required(VERSION 3.22.1)
project(openglesscope)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(OPENGLESSCOPE_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(OPENGLESSCOPE_LOCKED_GL_XML "${OPENGLESSCOPE_PROJECT_ROOT}/registry/gl.xml")
if(NOT EXISTS "${OPENGLESSCOPE_LOCKED_GL_XML}")
    message(FATAL_ERROR "Bundled canonical Khronos OpenGL registry snapshot is missing")
endif()
file(SHA256 "${OPENGLESSCOPE_LOCKED_GL_XML}" OPENGLESSCOPE_GL_XML_SHA256)
if(NOT OPENGLESSCOPE_GL_XML_SHA256 STREQUAL "b9ca2cfa5c676e901c20d34af3407f1687cde0f1336a5ff7a8974d04c7494ad3")
    message(FATAL_ERROR "Bundled OpenGL registry snapshot hash mismatch")
endif()
file(READ "${OPENGLESSCOPE_LOCKED_GL_XML}" OPENGLESSCOPE_GL_XML_TEXT)
foreach(OPENGLESSCOPE_REQUIRED_SENTINEL "GL_ES_VERSION_3_2" "GL_EXT_fragment_shading_rate" "GL_EXT_mesh_shader" "GL_ARM_shader_core_properties")
    string(FIND "${OPENGLESSCOPE_GL_XML_TEXT}" "${OPENGLESSCOPE_REQUIRED_SENTINEL}" OPENGLESSCOPE_SENTINEL_POS)
    if(OPENGLESSCOPE_SENTINEL_POS EQUAL -1)
        message(FATAL_ERROR "Bundled OpenGL registry is missing required sentinel ${OPENGLESSCOPE_REQUIRED_SENTINEL}")
    endif()
endforeach()
set(OPENGLESSCOPE_LOCKED_EGL_XML "${OPENGLESSCOPE_PROJECT_ROOT}/registry/egl.xml")
if(NOT EXISTS "${OPENGLESSCOPE_LOCKED_EGL_XML}")
    message(FATAL_ERROR "Bundled canonical Khronos EGL registry snapshot is missing")
endif()
file(SHA256 "${OPENGLESSCOPE_LOCKED_EGL_XML}" OPENGLESSCOPE_EGL_XML_SHA256)
if(NOT OPENGLESSCOPE_EGL_XML_SHA256 STREQUAL "3327619123cdaa999400b41a7060180616df42571a0fdd11e9726e026f6d0fb8")
    message(FATAL_ERROR "Bundled EGL registry snapshot hash mismatch")
endif()
file(READ "${OPENGLESSCOPE_LOCKED_EGL_XML}" OPENGLESSCOPE_EGL_XML_TEXT)
foreach(OPENGLESSCOPE_REQUIRED_EGL_SENTINEL "EGL_VERSION_1_5" "EGL_EXT_device_query" "EGL_EXT_device_enumeration" "EGL_EXT_surface_compression" "eglQuerySupportedCompressionRatesEXT")
    string(FIND "${OPENGLESSCOPE_EGL_XML_TEXT}" "${OPENGLESSCOPE_REQUIRED_EGL_SENTINEL}" OPENGLESSCOPE_EGL_SENTINEL_POS)
    if(OPENGLESSCOPE_EGL_SENTINEL_POS EQUAL -1)
        message(FATAL_ERROR "Bundled EGL registry is missing required sentinel ${OPENGLESSCOPE_REQUIRED_EGL_SENTINEL}")
    endif()
endforeach()
add_library(openglesscope SHARED openglesscope.cpp)
find_library(log-lib log)
find_library(egl-lib EGL)
find_library(gles-lib GLESv3)
target_link_libraries(openglesscope PRIVATE ${log-lib} ${egl-lib} ${gles-lib})
target_compile_options(openglesscope PRIVATE -Wall -Wextra -Werror -fstack-protector-strong -fvisibility=hidden)
target_link_options(openglesscope PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384" "-Wl,-z,relro" "-Wl,-z,now")
