cmake_minimum_required(VERSION 3.22.1)
project(openglesscope)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(openglesscope SHARED openglesscope.cpp)
find_library(log-lib log)
find_library(egl-lib EGL)
find_library(gles-lib GLESv3)
target_link_libraries(openglesscope PRIVATE ${log-lib} ${egl-lib} ${gles-lib})
target_compile_options(openglesscope PRIVATE -Wall -Wextra -Werror -fstack-protector-strong -fvisibility=hidden)
target_link_options(openglesscope PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384" "-Wl,-z,relro" "-Wl,-z,now")
